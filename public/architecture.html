/** Optional one-time provisioning. Reuses IDs already present in .env. */
import Anthropic from "@anthropic-ai/sdk";
import { existsSync, readFileSync, writeFileSync } from "node:fs";
if (
  !process.env.ANTHROPIC_API_KEY ||
  process.env.ANTHROPIC_API_KEY.startsWith("your_")
) {
  throw new Error("Set your own ANTHROPIC_API_KEY in .env first.");
}
const client = new Anthropic({ maxRetries: 0 });
let contents = existsSync(".env") ? readFileSync(".env", "utf8") : "";
function save(key: string, value: string) {
  const pattern = new RegExp("^\\s*" + key + "=.*$", "m");
  contents = pattern.test(contents)
    ? contents.replace(pattern, key + "=" + value)
    : contents.trimEnd() + "\n" + key + "=" + value + "\n";
  writeFileSync(".env", contents, { mode: 0o600 });
}
const valid = (value: string | undefined) =>
  Boolean(value && !value.startsWith("your_"));
if (!valid(process.env.CLAUDE_MANAGED_ENVIRONMENT_ID)) {
  const environment = await client.beta.environments.create({
    name: "C07 student environment",
    config: { type: "cloud", networking: { type: "limited" } },
  });
  // Save immediately so a later agent failure does not lose the environment ID.
  save("CLAUDE_MANAGED_ENVIRONMENT_ID", environment.id);
}
if (!valid(process.env.CLAUDE_MANAGED_AGENT_ID)) {
  const agent = await client.beta.agents.create({
    name: "C07 text-only student agent",
    model: process.env.CLAUDE_MODEL || "claude-opus-5-5",
    system: "Answer the student's question clearly and briefly.",
    tools: [],
    mcp_servers: [],
    skills: [],
  });
  save("CLAUDE_MANAGED_AGENT_ID", agent.id);
}
console.log(
  "Prerequisite IDs are saved in your local .env. Existing configured IDs were reused.",
);

      $("#session").textContent = value;
      status.textContent = "Observing live events…";
    } else if (type === "event") {
      const li = document.createElement("li");
      li.textContent = value;
      $("#events").append(li);
    } else if (type === "answer") {
      $("#answer").hidden = false;
      $("#answer").textContent += value;
    } else if (type === "reason") $("#reason").textContent = value;
    else if (type === "outcome") {
      finished = true;
      $("#outcome").textContent = value;
      status.textContent = "Observation finished.";
    } else if (type === "error") {
      failed = true;
      $("#outcome").textContent = "Error; completion not confirmed";
      status.textContent = value;
    }
  };
  try {
    const r = await fetch("/api/run", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ prompt }),
    });
    if (!r.ok) {
      const d = await r.json();
      throw new Error(d.error || "Request failed");
    }
    const reader = r.body.getReader(),
      decoder = new TextDecoder();
    let pending = "";
    while (true) {
      const { value, done } = await reader.read();
      pending += decoder.decode(value || new Uint8Array(), { stream: !done });
      const lines = pending.split("\n");
      pending = lines.pop();
      for (const line of lines) if (line) handle(JSON.parse(line));
      if (done) break;
    }
    if (pending.trim()) handle(JSON.parse(pending));
    if (!finished && !failed) {
      $("#outcome").textContent = "Incomplete";
      status.textContent = "Stream ended without a terminal outcome.";
    }
  } catch (e) {
    status.textContent = e.message;
    $("#outcome").textContent = "Error; completion not confirmed";
  } finally {
    button.disabled = false;
  }
});

3 · Understand the lesson-specific rules · part 1

Runs a real session and classifies its observed stop reason. The sample displays pauses; it does not implement approvals, resume or remote interrupt cleanup.

Source: src/outcome.ts · starting at line 1

export type Outcome =
  | "completed"
  | "interrupted"
  | "paused-budget"
  | "requires-action"
  | "error"
  | "terminated"
  | "incomplete";
export function classify(
  status: string,
  reason: string | undefined,
  interrupted: boolean,
): Outcome {
  if (status === "terminated") return "terminated";
  if (status !== "idle") return "incomplete";
  if (reason === "requires_action") return "requires-action";
  if (reason === "budget_reached") return "paused-budget";
  if (reason === "end_turn") return interrupted ? "interrupted" : "completed";
  return "error";
}

What the evidence proves

Runs a real session and classifies its observed stop reason. The sample displays pauses; it does not implement approvals, resume or remote interrupt cleanup.

Check the displayed outcome and source together. A rendered answer alone does not prove that a tool ran, a remote operation succeeded, or work was cancelled.
